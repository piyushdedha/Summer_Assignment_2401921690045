1. Valid Anagram 
class Solution(object):
    def isAnagram(self, s, t):
        if len(s) != len(t):
            return False
        freq = {}
        for ch in s:
            if ch in freq:
                freq[ch] += 1
            else:
                freq[ch] = 1
        for ch in t:
            if ch not in freq:
                return False
            freq[ch] -= 1
        for count in freq.values():
            if count != 0:
                return False
        return True


2. First Unique Character in a String 
class Solution:
    def firstUniqChar(self, s):
        freq = {}
        for ch in s:
            freq[ch] = freq.get(ch,0)+1
        for i in range(len(s)):
            if freq[s[i]]==1:
                return i
        return -1


3. Ransom Note 
class Solution:
    def canConstruct(self, ransomNote, magazine):
        letters={}
        for ch in magazine:
            letters[ch]=letters.get(ch,0)+1
        for ch in ransomNote:
            if ch not in letters or letters[ch]==0:
                return False
            letters[ch]-=1
        return True
